Q1 Kiểm Tra Lệch Giờ DB (Time Drift Window Check) DUAL / DATE ARITHMETIC
QUERY 1: Time Offset Verification 
SELECT 
    CASE WHEN ((TO_DATE(:ArgTransTime, 'yyyymmddhh24miss') - SYSDATE) * 24 * 60) > 20 
         THEN 'NG' ELSE 'OK' END AS v_Future,
    CASE WHEN ((TO_DATE(:ArgTransTime, 'yyyymmddhh24miss') - SYSDATE) * 24 * 60) < -20 
         THEN 'NG' ELSE 'OK' END AS V_past
FROM dual;
Tham số / Biểu thứcGiải thích giải phẫuÝ nghĩa thực tế trên sàn sản xuất
(Target_Date - SYSDATE) Phép trừ Date trong Oracle trả về số ngày (Real Number). Đo khoảng cách thời gian giữa máy EIF và máy chủ Database.
* 24 * 60 Đổi số ngày ra số phút (1 ngày = 24h = 1440 phút). Quy đổi về thang đo phút để so sánh với ngưỡng 20 phút.
Ngưỡng > 20 / < -20 Phát hiện máy trạm lệch giờ hoặc tin nhắn bị nghẽn (Lag). Chặn việc ghi nhận sai thứ tự thời gian các công đoạn sản xuất.

QUERY 2: Lot Verification & Left Outer Join 
SELECT a.lot_number, a.device_stype, a.device, a.operation,
       NVL(MAX(b.equipment_id), 'X') AS current_equip,
       MAX(a.in_hold) AS in_hold
  FROM ASFC_LOT_STATUS a,
       ASFC_EQPLOT_STATUS b
 WHERE a.plant = :gv_Plant AND a.lot_number = :gv_Lot
   AND a.operation = :ArgOper AND a.in_active = 'Y'
   AND b.plant(+) = a.plant AND b.lot_number(+) = a.lot_number
 GROUP BY a.lot_number, a.device_stype, a.device, a.operation;

Q3 CTE 4 Tầng Phân Cấp Tìm Rule Ưu Tiên (F_ILRULE_CHECK_SQL) HIERARCHY CASCADE
QUERY 3: 4-Tier Specificity Cascade CTE 
WITH RULE_DEVICE_GROUP AS (
    SELECT a.*, MIN(device_seq) OVER (PARTITION BY rule_id) AS min_deviceseq
      FROM (
        SELECT a.*, CASE WHEN rule_key_group1 = 'ALL' THEN 1 ELSE 0 END AS device_seq
          FROM BIZ_ILRULE_INFO a
         WHERE plant = :plant AND trans_code = :trans_code
           AND rule_key_group1 IN ('ALL', :deviceGroup)
           AND rule_type = :ArgType AND active_flag = 'Y'
      ) a
),
RULE_OPERATION_GROUP AS (
    SELECT b.*, MIN(oper_seq) OVER (PARTITION BY rule_id) AS min_operseq
      FROM (
        SELECT a.*, CASE WHEN rule_key_group2 = 'ALL' THEN 1 ELSE 0 END AS oper_seq
          FROM RULE_DEVICE_GROUP a
         WHERE rule_key_group2 IN ('ALL', :operation)
           AND device_seq = min_deviceseq
      ) b
),
RULE_EQPTYPE_GROUP AS (
    SELECT b.*, MIN(type_seq) OVER (PARTITION BY rule_id) AS min_typeseq
      FROM (
        SELECT a.*, CASE WHEN rule_key_group3 = 'ALL' THEN 1 ELSE 0 END AS type_seq
          FROM RULE_OPERATION_GROUP a
         WHERE rule_key_group3 IN ('ALL', :equipgroup)
           AND oper_seq = min_operseq
      ) b
)
SELECT rule_id, rule_key_group1, rule_key_group2, rule_key_group3, rule_key_group4,
       NVL(bind_item, 'NONE') AS bind_item, check_type, check_sql_1, check_sql_2,
       lot_abn_yn, equip_pause_yn, error_msg_e, rule_key
  FROM (
    SELECT c.*, MIN(equip_seq) OVER (PARTITION BY rule_id) AS min_equipseq
      FROM (
        SELECT a.*, CASE WHEN rule_key_group4 = 'ALL' THEN 1 ELSE 0 END AS equip_seq
          FROM RULE_EQPTYPE_GROUP a
         WHERE rule_key_group4 IN ('ALL', :equip_id)
           AND type_seq = min_typeseq
      ) c
  )
 WHERE equip_seq = min_equipseq
 ORDER BY rule_id, rule_key;
Q4 DENSE_RANK() Lùi Lịch Sử Tìm Công Đoạn Hoàn Thành Trước Đó F_PREOPER_INTERLOCK
QUERY 4: History Trace-back Window Function 
SELECT a.operation AS pre_oper, a.equip_type AS pre_equip_type,
       a.equip_group AS pre_equip_group, a.equipment_id AS pre_equip,
       b.group_value AS pre_oper_group
  FROM (
    SELECT equipment_id, operation, trans_time,
           MAX(CASE WHEN group_index = 12 THEN group_value END) AS equip_type,
           MAX(CASE WHEN group_index = 13 THEN group_value END) AS equip_group,
           DENSE_RANK() OVER (ORDER BY trans_time DESC, equipment_id) AS rank_no
      FROM asfc_eqplot_history a, adm_group_category_data b
     WHERE a.plant = :ArgPlant AND b.plant = a.plant
       AND b.group_target = 'EQUIPMENT' AND b.group_index IN (12, 13)
       AND b.group_object = a.equipment_id AND a.transaction = 'SPOU'
       AND a.lot_number = :ArgLot
       AND a.trans_time <= (
            SELECT MAX(trans_time) FROM asfc_eqplot_history
             WHERE plant = :ArgPlant AND lot_number = :ArgLot
               AND transaction = 'SPOU' AND trans_time < :ArgTransTime
               AND equipment_id IN (
                    SELECT group_object FROM adm_group_category_data
                     WHERE plant = :ArgPlant AND group_target = 'EQUIPMENT'
                       AND group_index = 13
                       AND group_value NOT IN (
                            SELECT code_name FROM adm_user_code_data
                             WHERE plant = :ArgPlant AND table_name = 'EIF_EQUIP_GROUP'
                               AND code_group1 = 'X' AND :v_HandlingPcb = 0
                       )
               )
       )
     GROUP BY equipment_id, operation, trans_time
  ) a, adm_group_category_data b
 WHERE b.plant = :ArgPlant AND b.group_target = 'OPERATION'
   AND b.group_index = 22 AND b.group_object = a.operation
   AND a.rank_no = 1;

SELECT NVL(MAX(pcb_id), 'N/A') AS uncompleted_pcb

    SELECT pcb_barcode AS pcb_id

QUERY 8: SMT Tool Life & Time Audit 
SELECT s.parts_lotno, s.trans_time,
       ROUND((SYSDATE - TO_DATE(s.trans_time, 'yyyymmddhh24miss')) * 24, 2) AS usage_hours,
       NVL(t.life_time, 0) AS cur_stroke, NVL(t.alarm_period, 0) AS limit_stroke,
       CASE 
           WHEN ROUND((SYSDATE - TO_DATE(s.trans_time, 'yyyymmddhh24miss')) * 24, 2) >= 4 
                THEN 'NG: Khung in gá quá 4 giờ'
           WHEN NVL(t.life_time, 0) >= NVL(t.alarm_period, 0) AND t.expire_check_flag = 'Y' 
                THEN 'NG: Vượt số lần dập quy định'
           ELSE 'OK'
       END AS status
  FROM etc_erp_eqpparts_status s
  LEFT JOIN amat_tools_item t 
    ON t.plant = s.plant AND t.parts_stype = 'T0004' AND t.parts_itemid = s.parts_lotno
 WHERE s.plant = :ArgPlant AND s.parts_kind = 'T0004'
   AND s.equipment_id = :v_MainEquip AND s.itemcode = 'T0004';
Q9 Thuật Toán Phễu 5 Tầng Đối Chiếu Lỗi 4M Vật Tư (F_MAT_4MABN_CHECK) ABN_4M_STATUS
QUERY 9: 5-Tier Material 4M Audit 
WITH ABN_ITEM_CODE AS (
    SELECT a.* FROM ABN_4M_STATUS a
     WHERE plant = :ArgPlant AND abn_type = :ArgType AND abn_key_1 = :ArgItemCode
),
ABN_ITEM_LOT AS (
    SELECT b.*, MIN(ItemLot_Seq) OVER (PARTITION BY abn_type) AS min_itemlot_seq
      FROM (
        SELECT a.*, CASE WHEN abn_key_2 = 'ALL' THEN 1 ELSE 0 END AS ItemLot_Seq
          FROM ABN_ITEM_CODE a WHERE abn_key_2 IN ('ALL', :ArgItemLot)
      ) b
),
ABN_PROD_DATE AS (
    SELECT b.*, MIN(ProdDate_Seq) OVER (PARTITION BY abn_type) AS min_ProdDate_Seq
      FROM (
        SELECT a.*, CASE WHEN abn_key_3 = 'ALL' THEN 1 ELSE 0 END AS ProdDate_Seq
          FROM ABN_ITEM_LOT a WHERE abn_key_3 IN ('ALL', :ArgPdate) AND ItemLot_Seq = min_itemlot_seq
      ) b
),
ABN_REEL_NO AS (
    SELECT b.*, MIN(ReelNo_Seq) OVER (PARTITION BY abn_type) AS min_ReelNo_Seq
      FROM (
        SELECT a.*, CASE WHEN abn_key_4 = 'ALL' THEN 1 ELSE 0 END AS ReelNo_Seq
          FROM ABN_PROD_DATE a WHERE abn_key_4 IN ('ALL', :ArgReelNo) AND ProdDate_Seq = min_ProdDate_Seq
      ) b
)
SELECT NVL(MIN(ABN_ID), 'X') AS final_abn_id
  FROM (
    SELECT b.*, MIN(vendor_seq) OVER (PARTITION BY abn_type) AS min_vendor_seq
      FROM (
        SELECT a.*, CASE WHEN abn_key_5 = 'ALL' THEN 1 ELSE 0 END AS vendor_seq
          FROM ABN_REEL_NO a WHERE abn_key_5 IN ('ALL', :ArgVendorName) AND ReelNo_Seq = min_ReelNo_Seq
      ) b
  )
 WHERE vendor_seq = min_vendor_seq;

-- Đo thời gian xử lý chính xác đến một phần nghìn giây (Process Time in Seconds)
:v_ProcTime := ROUND((TO_DATE(TO_CHAR(SYSDATE, 'yyyymmddhh24miss'), 'yyyymmddhh24miss') 
                     - TO_DATE(:gv_StartTime, 'yyyymmddhh24miss')) * 24 * 60 * 60, 3);
